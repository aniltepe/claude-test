V=/var/jb/var/mobile/py-nba/venv
CSO="$V/lib/python3.9/site-packages/_cffi_backend.cpython-39-darwin.so"
vtool -set-build-version ios 15.0 15.0 -replace -output "$CSO" "$CSO"
ldid -S "$CSO"
otool -l "$CSO" | grep -A3 LC_BUILD_VERSION
"$V/bin/python3" -c "import cffi; print(cffi.__version__)"



cat > ~/bin/retarget-ios.py <<'EOF'
import struct, sys

path = sys.argv[1]
d = bytearray(open(path, "rb").read())

magic = struct.unpack_from("<I", d, 0)[0]
if magic != 0xfeedfacf:
    sys.exit(f"not a 64-bit little-endian Mach-O (magic {magic:#x})")

ncmds = struct.unpack_from("<I", d, 16)[0]
off, changed = 32, 0

for _ in range(ncmds):
    cmd, cmdsize = struct.unpack_from("<II", d, off)
    if cmd == 0x32:                       # LC_BUILD_VERSION
        plat = struct.unpack_from("<I", d, off + 8)[0]
        if plat != 2:
            struct.pack_into("<I", d, off + 8, 2)     # 2 = iOS
            changed += 1
    elif cmd == 0x24:                     # LC_VERSION_MIN_MACOSX
        struct.pack_into("<I", d, off, 0x25)          # LC_VERSION_MIN_IPHONEOS
        changed += 1
    off += cmdsize

if not changed:
    sys.exit("nothing to change (already iOS?)")

open(path, "wb").write(d)
print(f"patched {changed} load command(s) in {path}")
EOF
chmod +x ~/bin/retarget-ios.py


cp "$SO" "$SO.bak"
python3 ~/bin/retarget-ios.py "$SO"
ldid -S "$SO"
otool -l "$SO" | grep -A3 -E "LC_BUILD_VERSION|LC_VERSION_MIN"
"$V/bin/python3" -c "import cffi; print(cffi.__version__)"
